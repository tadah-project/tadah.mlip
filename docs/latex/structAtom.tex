\doxysubsection{Detailed Description}
Container to represent atom properties

Usage example\+: \begin{DoxyVerb}# create an empty atom object - all attributes are left uninitialised
Atom atom;

# set atom symbol
atom.symbol="Ti"

# set atom name, atomic number... see Element class for attributes

# set atom position to (1.1, 2.2, 3.3)
atom.position(0) = 1.1;
atom.position(1) = 2.2;
atom.position(2) = 3.3;

# set force
atom.force(0)= 0.1;
atom.force(1)= -0.2;
atom.force(2)= 0.3;
\end{DoxyVerb}


Usage example\+: \begin{DoxyVerb}# Use constructor to fully initialise this object
# Positions and forces as above
Element element = PeriodicTable().find_by_symbol("Ti");
Atom atom(element, 1.1, 2.2, 3.3, 0.1, -0.2, 0.3);
\end{DoxyVerb}


Usage example\+: \begin{DoxyVerb}# Print atom object using streams:
std::cout << atom;

# Print position only
std::cout << atom.position
\end{DoxyVerb}
