304  nweights.resize(weights.rows());

306  nweights(0) = weights(0);
307  for (size_t i=1; i<weights.size(); ++i) {
308

310  nweights(i) = weights(i) / norm.std_dev(i);
311  else
312  nweights(i) = weights(i);
313 
314  nweights(0) -= norm.mean(i)*nweights(i);

316  }

352  StructureDB predict(Config config_pred, StructureDB &stdb, DC_Base &dc,
353  t_type &predicted_error) {
354 
355  LinearRegressor::read_sigma(config_pred,Sigma);
356  DesignMatrix<BF> dm(bf,config_pred);
357  dm.scale=false; // do not scale energy, forces and stresses
358  dm.build(stdb,norm,dc);
359 
360  phi_type &phi = dm.Phi;
361 
362  // compute error
363  predicted_error = (phi*Sigma*phi.transpose()).diagonal();
364  double beta = config.get<double>("BETA");
365  predicted_error.array() += 1.0/beta;
366 
367  // compute energy, forces and stresses
368  t_type Tpred = phi*weights;
369 
370  // Construct StructureDB object with predicted values
371  StructureDB stdb_;
372  stdb_.structures.resize(stdb.size());
373  size_t i=0;
374  for (size_t s=0; s<stdb.size(); ++s) {
375  stdb_(s) = Structure(stdb(s));
376 
377  predicted_error(i) = sqrt(predicted_error(i));
378 
379  stdb_(s).energy = Tpred(i++);
380  if (config_pred.get<bool>("FORCE")) {
381  for (size_t a=0; a<stdb(s).natoms(); ++a) {
382  for (size_t k=0; k<3; ++k) {
383  predicted_error(i) = sqrt(predicted_error(i));
384  stdb_(s).atoms[a].force[k] = Tpred(i++);
385  }
386  }
387  }
388  if (config_pred.get<bool>("STRESS")) {
389  for (size_t x=0; x<3; ++x) {
390  for (size_t y=x; y<3; ++y) {
391  predicted_error(i) = sqrt(predicted_error(i));
392  stdb_(s).stress(x,y) = Tpred(i++);
393  if (x!=y)
394  stdb_(s).stress(y,x) = stdb_(s).stress(x,y);
395  }
396  }
397  }
398  }
399  return stdb_;
