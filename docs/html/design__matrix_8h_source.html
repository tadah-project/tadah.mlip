199  void compute_stdevs(const StructureDB &stdb) {
200  Eigen::ArrayXd evec(stdb.size());
201  Eigen::ArrayXXd svec(6,stdb.size());
202  size_t num_forces=0;
203  for (size_t s=0; s<stdb.size(); ++s) {
204  evec(s) = stdb(s).energy/stdb(s).natoms();
205  if (stress) {
206  size_t j=0;
207  for (size_t x=0; x<3; ++x) {
208  for (size_t y=x; y<3; ++y) {
209  svec(j++,s)=stdb(s).stress(x,y);
210  }
211  }
212  }
213  num_forces +=3*stdb(s).natoms();
214  }
215 
216  e_std_dev = std::sqrt((evec - evec.mean()).square().sum()/(evec.size()-1));
217  if (verbose) std::cout << "Energy standard deviation (eV/atom): " << e_std_dev << std::endl;
218 
219  if (stress) {
220  for (size_t j=0; j<6; ++j) {
221  s_std_dev[j] = std::sqrt((svec.row(j) - svec.row(j).mean()).square().sum()/(svec.row(j).size()-1));
222  if (verbose) std::cout << "Stress standard deviation (eV): " << s_std_dev[j] << std::endl;
223  }
224  }
225 
226 
227  if (force) {
228  size_t j=0;
229  Eigen::ArrayXd fvec(num_forces);
230  for (size_t s=0; s<stdb.size(); ++s) {
231  for (const Atom &a : stdb(s).atoms) {
232  fvec(j++) = a.force(0);
233  fvec(j++) = a.force(1);
234  fvec(j++) = a.force(2);
235  }
236  }
237  //fvec /= e_std_dev;
238  //svec /= e_std_dev;
239  // e_std_dev has units of energy
240  // f_std_dev has units of inverse distance
241  f_std_dev = std::sqrt((fvec - fvec.mean()).square().sum()/(fvec.size()-1));
242  if (verbose) std::cout << "Force standard deviation (1/A): " << f_std_dev << std::endl;
243  }
