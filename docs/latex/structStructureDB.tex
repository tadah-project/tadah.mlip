\doxysubsection{Detailed Description}
Container for a collection of \mbox{\hyperlink{structStructure}{Structure(s)}}.

Usage example\+: \begin{DoxyVerb}# Default constructor creates an empty StructureDB object.
StructureDB stdb;
# Add structures from the file "db.dat"
stdb.add("db.dat");
\end{DoxyVerb}


Usage example\+: \begin{DoxyVerb}# Load all structure files listed in a Config file
Config config("config");
StructureDB stdb(config);
\end{DoxyVerb}


Usage example\+: \begin{DoxyVerb}# Print this object summary using streams:
std::cout << stdb;

# Print 3rd Atom data from the 1st Structure
std::cout << stdb(0,2);
\end{DoxyVerb}
