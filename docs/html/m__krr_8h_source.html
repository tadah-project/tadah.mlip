297  void convert_to_nweights() {
298  if (kernel.get_label()!="Kern_Linear") {
299  //kernalize weights
300  kweights = ekm.KK.transpose()*weights;
301  }
302  else if(config.get<bool>("NORM") && kernel.get_label()=="Kern_Linear") {
303  // normalise weights such that when predict is called
304  // we can supply it with a non-normalised descriptor
305  kweights.resize(weights.rows());
306  kweights(0) = weights(0);
307  for (size_t i=1; i<weights.size(); ++i) {
308 
309  if (norm.std_dev(i) > std::numeric_limits<double>::min())
310  kweights(i) = weights(i) / norm.std_dev(i);
311  else
312  kweights(i) = weights(i);
313 
314  kweights(0) -= norm.mean(i)*kweights(i);
315 
316  }
317  }
318  else {
319  kweights=weights;
320  }
321  }

323  void convert_to_weights() {
324  if(config.get<bool>("NORM") && kernel.get_label()=="Kern_Linear") {
325  // convert normalised weights back to "normal"
326  weights.resize(kweights.rows());
327  weights.setZero();
328  weights(0) = kweights(0);
329  for (size_t i=1; i<kweights.size(); ++i) {
330 
331  if (norm.std_dev(i) > std::numeric_limits<double>::min())
332  weights(i) = kweights(i) * norm.std_dev(i);
333  else
334  weights(i) = kweights(i);
335 
336  weights(0) += kweights(i)*norm.mean(i);
337 
338  }
339  }
340  else {

344  StructureDB predict(Config config_pred, StructureDB &stdb, DC_Base &dc,
345  t_type &predicted_error) {
346 
347  LinearRegressor::read_sigma(config_pred,Sigma);
348  DesignMatrix<K> dm(kernel,config_pred);
349  dm.scale=false; // do not scale energy, forces and stresses
350  dm.build(stdb,norm,dc);
351 
352  phi_type &phi = dm.Phi;
353 
354  // compute error
355  predicted_error = (phi*Sigma*phi.transpose()).diagonal();
356  double beta = config.get<double>("BETA");
357  predicted_error.array() += 1.0/beta;
358 
359  // compute energy, forces and stresses
360  t_type Tpred = phi*weights;
361 
362  // Construct StructureDB object with predicted values
363  StructureDB stdb_;
364  stdb_.structures.resize(stdb.size());
365  size_t i=0;
366  for (size_t s=0; s<stdb.size(); ++s) {
367  stdb_(s) = Structure(stdb(s));
368 
369  predicted_error(i) = sqrt(predicted_error(i));
370 
371  stdb_(s).energy = Tpred(i++);
372  if (config_pred.get<bool>("FORCE")) {
373  for (size_t a=0; a<stdb(s).natoms(); ++a) {
374  for (size_t k=0; k<3; ++k) {
375  predicted_error(i) = sqrt(predicted_error(i));
376  stdb_(s).atoms[a].force[k] = Tpred(i++);
377  }
378  }
379  }
380  if (config_pred.get<bool>("STRESS")) {
381  for (size_t x=0; x<3; ++x) {
382  for (size_t y=x; y<3; ++y) {
383  predicted_error(i) = sqrt(predicted_error(i));
384  stdb_(s).stress(x,y) = Tpred(i++);
385  if (x!=y)
386  stdb_(s).stress(y,x) = stdb_(s).stress(x,y);
387  }
388  }
389  }
390  }
391  return stdb_;
