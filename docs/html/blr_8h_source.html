261  void train(phi_type &Phi, t_type &T) {
262  if(trained) throw std::runtime_error("This object is already trained!");
263 
264 
265  // do training...
266  // In general there are 3 methods to solve least squares problem:
267  // 1) Normal equation - fast but least accurate:
268  //Eigen::MatrixXd I = Eigen::MatrixXd::Identity(Phi.cols(),Phi.cols());
269  //Eigen::Matrix<double, Eigen::Dynamic, Eigen::Dynamic> PhiPhi = Phi.transpose()*Phi;
270  //weights = (PhiPhi+lambda*I).inverse() * Phi.transpose() * T;
271  // 2) QR - in between 1) and 3) in therm of speed and accuracy
272  //weights = Phi.colPivHouseholderQr().solve(T);
273  // 3) SVD - most accurate but the slowest
274  // w = V*D*S(*)*U^T*T
275  // where S(*) = S.asDiagonal().inverse()
276  // and D is a filter matrix with diagonal entires: S(i)^2/(S(i)^2+lambda)
277  using mat=Eigen::MatrixXd;
278 
279  lambda=config.get<double>("LAMBDA");
280  if (lambda == 0)
281  weights = Phi.bdcSvd(Eigen::ComputeThinU | Eigen::ComputeThinV).solve(T);
282  else {
283  /*
284  using mat=Eigen::MatrixXd;
285  using vec=Eigen::VectorXd;
286  mat PhiTPhi = Phi.transpose()*Phi;
287  Eigen::EigenSolver<mat> es(PhiTPhi);
288  vec U = es.eigenvalues().real();
289  mat V = es.eigenvectors().real();
290  mat A = mat::Identity(Phi.cols(),Phi.cols());
291  for (size_t i=0; i<Phi.cols();++i)
292  A(i,i) = 1.0/(U(i)+lambda);
293  weights=V*A*V.transpose()*Phi.transpose()*T;
294  */
295  //Eigen::JacobiSVD<mat> svd(Phi, Eigen::ComputeThinU | Eigen::ComputeThinV);
296  Eigen::BDCSVD<mat> svd(Phi, Eigen::ComputeThinU | Eigen::ComputeThinV);
297  mat S =svd.singularValues();
298  const mat &U =svd.matrixU();
299  const mat &V =svd.matrixV();
300  mat S_inv = S.cwiseInverse();
301 
302  // Filter out v.small singularvalues from S and its inverse
303  for (size_t i=0; i<Phi.cols(); ++i) {
304  if (S(i,0)<std::numeric_limits<double>::min()) {
305  S_inv(i,0) = 0.0;
306  S(i,0) = 0.0;
307  }
308  }
309 
310  if (lambda<0) {
311  EA ea(config);
312  ea.run(Phi,T,S,S_inv,U,V,lambda);
313  }
314  mat D(Phi.cols(),Phi.cols());
315  D.setZero();
316  for (size_t i=0; i<Phi.cols(); ++i) {
317  double s = S(i,0);
318  if (s>std::numeric_limits<double>::min())
319  D(i,i) = (s*s)/(s*s+lambda);
320  else
321  D(i,i) = 0.0;
322  }
323  if (verbose) std::cout << "REG LAMBDA: " << lambda << std::endl;
324  weights = V*D*S_inv.asDiagonal()*U.transpose()*T;
325  //weights = V*S_inv.asDiagonal()*U.transpose()*T;
326  //weights = V*S.asDiagonal().inverse()*U.transpose()*T;
327  }
328  if(config.get<bool>("NORM") && BF::get_label()=="Linear") {
329  // normalise weights such that when predict is called
330  // we can supply it with a non-normalised descriptor
331  nweights.resize(weights.rows());
332  nweights(0) = weights(0);
333  for (size_t i=1; i<weights.size(); ++i) {
334 
335  if (norm.std_dev(i) > std::numeric_limits<double>::min())
336  nweights(i) = weights(i) / norm.std_dev(i);
337  else
338  nweights(i) = weights(i);
339 
340  nweights(0) -= norm.mean(i)*nweights(i);
341 
342  }
343  }
344  else {
345  nweights=weights;
346  }
347  trained=true;
348  }
