41  //std::cout << "COLS NORM: " << nn << std::endl;
42  // compute mean and st_dev
43  for (size_t i=0; i<nn; ++i) {
44  Eigen::VectorXd v(n);
45  size_t b=0;
46  for (size_t s=0; s<st_desc_db.size(); ++s) {
47  for (const auto &aed:st_desc_db(s).aed) {
48  v(b++) = aed(i);
49  }
50  }
51  std_dev(i) = std::sqrt((v.array() - v.mean()).square().sum()/(v.size()));
52  mean(i) = v.mean();
53  }
54  if (verbose) std::cout << std::endl << "NORMALISER STDEV : " << std_dev.transpose() << std::endl;
55  if (verbose) std::cout << std::endl << "NORMALISER MEAN :" << mean.transpose() << std::endl;
