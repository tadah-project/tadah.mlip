302  // do training...
303  // In general there are 3 methods to solve least squares problem:
304  // 1) Normal equation - fast but least accurate:
305  //Eigen::MatrixXd I = Eigen::MatrixXd::Identity(Phi.cols(),Phi.cols());
306  //Eigen::Matrix<double, Eigen::Dynamic, Eigen::Dynamic> PhiPhi = Phi.transpose()*Phi;
307  //weights = (PhiPhi+lambda*I).inverse() * Phi.transpose() * T;
308  // 2) QR - in between 1) and 3) in therm of speed and accuracy
309  //weights = Phi.colPivHouseholderQr().solve(T);
310  // 3) SVD - most accurate but the slowest
311  // w = V*D*S(*)*U^T*T
312  // where S(*) = S.asDiagonal().inverse()
313  // and D is a filter matrix with diagonal entires: S(i)^2/(S(i)^2+lambda)
314  using mat=Eigen::MatrixXd;
315 
316  lambda=config.get<double>("LAMBDA");
317  if (lambda == 0)
318  weights = Phi.bdcSvd(Eigen::ComputeThinU | Eigen::ComputeThinV).solve(T);
319  else {
320  //Eigen::JacobiSVD<mat> svd(Phi, Eigen::ComputeThinU | Eigen::ComputeThinV);
321  Eigen::BDCSVD<mat> svd(Phi, Eigen::ComputeThinU | Eigen::ComputeThinV);
322  mat S =svd.singularValues();
323  const mat &U =svd.matrixU();
324  const mat &V =svd.matrixV();
325  mat S_inv = S.cwiseInverse();
326 
327 
328  // Filter out v.small singularvalues from S and its inverse
329  for (size_t i=0; i<Phi.cols(); ++i) {
330  if (S(i,0)<std::numeric_limits<double>::min()) {
331  S_inv(i,0) = 0.0;
332  S(i,0) = 0.0;
333  }
334  }
335 
336  if (lambda<0) {

339  }
340  mat D(Phi.cols(),Phi.cols());
341  D.setZero();
342  for (size_t i=0; i<Phi.cols(); ++i) {
343  double s = S(i,0);
344  if (s>std::numeric_limits<double>::min())
345  D(i,i) = (s*s)/(s*s+lambda);
346  else
347  D(i,i) = 0.0;
348  }
